def _get_consecutive_frames(world4d, center_frame, n=10, step=1):
    """Return up to n frame keys from world4d centered on center_frame, with stride step."""
    sorted_keys = sorted(world4d.keys())
    if center_frame not in sorted_keys:
        return [center_frame]
    ci = sorted_keys.index(center_frame)
    half = (n * step) // 2
    start = max(0, ci - half)
    end = min(len(sorted_keys), start + n * step)
    start = max(0, end - n * step)
    return sorted_keys[start:end:step]


def generate_worst_frames_pdf(
    world4d_fused,
    final_w1,
    final_w2,
    err_fused,
    cam1_name,
    cam2_name,
    output_dir,
    video_base_path,
    output_folder_name,
    top_percentile=10,
    m=10,
    seq_len=10,
    n_subsample=10,
):
    """
    Build a separate PDF for the M worst-error frames, selected via Gonzalez
    greedy max-spread sampling from the top-percentile error pool.

    For each selected frame the PDF contains three pages showing seq_len consecutive
    frames centred on the selected frame:
      1. 4×seq_len RGB cross-projection strip
           row 1: cam1 own pose (cyan)
           row 2: cam2 own pose (orange)
           row 3: cam1 frame + cam2 cross-projected (magenta)
           row 4: cam2 frame + cam1 cross-projected (magenta)
      2. 2×seq_len 3D zoomed-in skeleton comparison (no align / procrustes)
      3. 2×seq_len 3D full-court view (no align / procrustes)

    The PDF also opens with an error histogram showing the selection threshold.

    Args:
        world4d_fused: fused world4d dict (contains per-camera RGB and joint data)
        final_w1, final_w2: per-view world4d dicts (used for 3D views)
        err_fused: error dict from compute_fused_errors
        cam1_name, cam2_name: display labels
        output_dir: directory to save the PDF
        video_base_path: root path for original video files
        output_folder_name: base name for the PDF file
        top_percentile: pool = frames in the top N% by PA-MPJPE error
        m: number of frames to select via Gonzalez sampling
        seq_len: number of consecutive frames to show per selected pose
    """
    metric = 'pose_l2_procrustes_mean'
    valid_mask = err_fused['has_match'] & ~np.isnan(err_fused[metric])
    valid_indices = err_fused['frame_idx'][valid_mask]
    valid_errors = err_fused[metric][valid_mask]

    if len(valid_indices) == 0:
        print("[Worst Frames PDF] No valid frames, skipping.")
        return

    # --- page 1: histogram ---
    fig_hist = plot_error_histogram_worst_frames(
        err_fused, metric=metric, top_percentile=top_percentile,
        title=f"PA-MPJPE Distribution — top {top_percentile}% threshold (red)"
    )

    # --- pool B: top-percentile frames ---
    threshold = np.percentile(valid_errors, 100 - top_percentile)
    bad_mask = valid_errors >= threshold
    bad_indices = valid_indices[bad_mask].tolist()
    bad_errors = valid_errors[bad_mask].tolist()
    frame_to_error = dict(zip(bad_indices, bad_errors))

    if not bad_indices:
        print("[Worst Frames PDF] Pool B is empty, skipping.")
        plt.close(fig_hist)
        return

    # --- extract pose features for Gonzalez (flattened world joints of first person) ---
    feat_list, idx_list, err_list = [], [], []
    for f, e in zip(bad_indices, bad_errors):
        fd = world4d_fused.get(f, {})
        cam1_data = fd.get('camera_1_data') or fd.get('camera_2_data')
        if cam1_data:
            jw = np.array(cam1_data[0]['joints_world'])
            feat_list.append(jw.flatten())
            idx_list.append(f)
            err_list.append(e)

    if not feat_list:
        feat_list = [[e] for e in bad_errors]
        idx_list = bad_indices
        err_list = bad_errors

    features = np.array(feat_list)
    sort_order = np.argsort(err_list)[::-1]
    sorted_indices = [idx_list[i] for i in sort_order]
    sorted_features = features[sort_order]

    selected_frames = gonzalez_sample_frames(sorted_indices, sorted_features, m)
    print(f"[Worst Frames PDF] Selected {len(selected_frames)} frames: {selected_frames}")

    # --- build figure list ---
    figures = [(f"Error Distribution (top {top_percentile}% shaded red)", fig_hist)]

    for i, f in enumerate(selected_frames):
        err_val = frame_to_error.get(f, float('nan'))
        seq_label = f"Frame {f} | PA-MPJPE = {err_val:.4f} m"
        prefix = f"Bad Frame {i+1}/{len(selected_frames)}"

        # Get seq_len frames centred on f, subsampled by n_subsample
        seq_fused = _get_consecutive_frames(world4d_fused, f, seq_len, step=n_subsample)
        seq_w1    = _get_consecutive_frames(final_w1,       f, seq_len, step=n_subsample)
        seq_w2    = _get_consecutive_frames(final_w2,       f, seq_len, step=n_subsample)

        # 4 rows × seq_len RGB cross-projection strip
        fig_rgb = visualize_worst_frames_rgb_strip(
            world4d_fused, seq_fused,
            cam1_name=cam1_name, cam2_name=cam2_name,
            video_base_path=video_base_path,
            title=f"{prefix} — RGB Strip | {seq_label}",
        )
        if fig_rgb is not None:
            figures.append((f"{prefix} RGB Strip", fig_rgb))

        # 4 rows × seq_len RGB strip cropped around joint projections
        fig_rgb_crop = visualize_worst_frames_rgb_strip(
            world4d_fused, seq_fused,
            cam1_name=cam1_name, cam2_name=cam2_name,
            video_base_path=video_base_path,
            title=f"{prefix} — RGB Strip Cropped | {seq_label}",
            zoom_in=True,
        )
        if fig_rgb_crop is not None:
            figures.append((f"{prefix} RGB Strip Cropped", fig_rgb_crop))

        # 4 rows × seq_len RGB strip cropped, cam2-centered cross-projection
        fig_rgb_crop_c2 = visualize_worst_frames_rgb_strip_cam2centered(
            world4d_fused, seq_fused,
            cam1_name=cam1_name, cam2_name=cam2_name,
            video_base_path=video_base_path,
            title=f"{prefix} — RGB Strip Cropped Cam2-Centered | {seq_label}",
            zoom_in=True,
        )
        if fig_rgb_crop_c2 is not None:
            figures.append((f"{prefix} RGB Strip Cropped Cam2-Centered", fig_rgb_crop_c2))

        # 2 rows × seq_len zoomed 3D skeletons
        fig_zoom = visualize_frame_comparison_grid_zoomed(
            final_w1, final_w2, seq_w1,
            cam1_name=cam1_name, cam2_name=cam2_name,
            title=f"{prefix} — 3D Zoomed | {seq_label}",

def _to_numpy_shape(shape_tensor, idx):
    """Extract shape row idx as a (10,) float32 numpy array."""
    s = shape_tensor[idx]
    if hasattr(s, 'cpu'):
        return s.detach().cpu().float().numpy().flatten()
    return np.asarray(s, dtype=np.float32).flatten()


def _get_chunk_start_for_frame(frame, chunk_size, chunk_step):
    """Return the owning PromptHMR chunk's start frame index for a world4d frame.

    Uses the stored 'chunk_start' key when available (set by fuse_results.py).
    Falls back to inferring from 'og_rgb_frame' for pkl files created before
    that key was added. Returns None if neither field is available.
    """
    import math
    cs = frame.get('chunk_start', None)
    if cs is not None:
        return int(cs)
    og = frame.get('og_rgb_frame', None)
    if og is None:
        return None
    n = max(0, math.ceil((og - chunk_size + 1) / chunk_step))
    return n * chunk_step


def compute_beta_consistency(final_w1, final_w2):
    """Compute L2 distance between the shape (beta) estimates from each camera for each PID.

    After match_pid_ids has aligned PIDs across cameras, the first frame where
    a PID appears in both cameras gives us both cameras' beta estimate for that
    person.  Shape is (roughly) constant within a video so one sample suffices.

    Returns:
        dict {int pid: float l2}
    """
    beta_l2_by_pid = {}
    for global_idx in sorted(final_w1.keys()):
        if global_idx not in final_w2:
            continue
        frame1 = final_w1[global_idx]
        frame2 = final_w2[global_idx]
        if len(frame1.get('track_id', [])) == 0 or len(frame2.get('track_id', [])) == 0:
            continue
        pids1 = frame1['track_id'].numpy().astype(int) if isinstance(frame1['track_id'], torch.Tensor) else np.array(frame1['track_id']).astype(int)
        pids2 = frame2['track_id'].numpy().astype(int) if isinstance(frame2['track_id'], torch.Tensor) else np.array(frame2['track_id']).astype(int)
        for pid in sorted(set(pids1) & set(pids2)):
            if pid in beta_l2_by_pid:
                continue
            i1 = int(np.where(pids1 == pid)[0][0])
            i2 = int(np.where(pids2 == pid)[0][0])
            s1 = _to_numpy_shape(frame1['shape'], i1)
            s2 = _to_numpy_shape(frame2['shape'], i2)
            beta_l2_by_pid[int(pid)] = float(np.linalg.norm(s1 - s2))
        if len(beta_l2_by_pid) > 0 and all(p in beta_l2_by_pid for p in sorted(set(pids1) & set(pids2))):
            # early exit once all currently-seen PIDs are recorded
            pass
    return beta_l2_by_pid


def collect_chunk_shapes(world4d, chunk_size=500, chunk_step=250):
    """For each PID in world4d, collect the shape (beta) estimate per PromptHMR chunk.

    Returns:
        dict {int pid: {int chunk_start: np.ndarray shape (10,)}}
    """
    chunk_shapes = {}
    for frame_data in world4d.values():
        cs = _get_chunk_start_for_frame(frame_data, chunk_size, chunk_step)
        if cs is None:
            continue
        if len(frame_data.get('track_id', [])) == 0:
            continue
        pids = frame_data['track_id'].numpy().astype(int) if isinstance(frame_data['track_id'], torch.Tensor) else np.array(frame_data['track_id']).astype(int)
        for i, pid in enumerate(pids):
            pid = int(pid)
            if pid not in chunk_shapes:
                chunk_shapes[pid] = {}
            if cs not in chunk_shapes[pid]:
                chunk_shapes[pid][cs] = _to_numpy_shape(frame_data['shape'], i)
    return chunk_shapes


def compute_chunk_degradation(final_w1, final_w2, fps, chunk_size=500, chunk_step=250):
    """Compute pairwise cross-camera error as a function of position within a PromptHMR chunk.

    For each frame pair in (final_w1, final_w2) where og_rgb_frame is available,
    records the position within the owning chunk and the pairwise error.  Averages
    across all chunks at each position.

    Returns:
        dict with keys t_sec, trans_l2_mean, trans_l2_std,
                       pose_l2_mean, pose_l2_std,
                       mpjpe_mean, mpjpe_std, n_samples
        (all values are np.ndarrays of the same length)
    """
    from collections import defaultdict
    errors_by_pos = defaultdict(list)

    for global_idx in sorted(final_w1.keys()):
        if global_idx not in final_w2:
            continue
        frame1 = final_w1[global_idx]
        frame2 = final_w2[global_idx]

        og_frame = frame1.get('og_rgb_frame', None)
        if og_frame is None:
            continue
        cs = _get_chunk_start_for_frame(frame1, chunk_size, chunk_step)
        if cs is None:
            continue
        pos = int(og_frame) - int(cs)
        if pos < 0 or pos >= chunk_size:
            continue

        m = extract_matched_joints_by_pid_single(frame1, frame2)
        if m is None:
            continue
        e = compute_frame_errors(*m)
        errors_by_pos[pos].append({
            'trans_l2': e.get('trans_l2', np.nan),
            'pose_l2': e.get('pose_l2_procrustes', np.nan),
            'mpjpe': e.get('mpjpe', np.nan),
        })

    if not errors_by_pos:
        return {}

    positions = sorted(errors_by_pos.keys())
    result = {
        't_sec': np.array([p / fps for p in positions]),
        'trans_l2_mean': np.array([np.nanmean([x['trans_l2'] for x in errors_by_pos[p]]) for p in positions]),
        'trans_l2_std':  np.array([np.nanstd( [x['trans_l2'] for x in errors_by_pos[p]]) for p in positions]),
        'pose_l2_mean':  np.array([np.nanmean([x['pose_l2']   for x in errors_by_pos[p]]) for p in positions]),
        'pose_l2_std':   np.array([np.nanstd( [x['pose_l2']   for x in errors_by_pos[p]]) for p in positions]),
        'mpjpe_mean':    np.array([np.nanmean([x['mpjpe']      for x in errors_by_pos[p]]) for p in positions]),
        'mpjpe_std':     np.array([np.nanstd( [x['mpjpe']      for x in errors_by_pos[p]]) for p in positions]),
        'n_samples':     np.array([len(errors_by_pos[p]) for p in positions]),
    }
    return result

def extract_per_pid_shapes(final_w1, final_w2):
    """Return {pid: {'cam1': np.array(10,), 'cam2': np.array(10,)}} for matched PIDs."""
    shapes = {}
    for global_idx in sorted(final_w1.keys()):
        if global_idx not in final_w2:
            continue
        frame1 = final_w1[global_idx]
        frame2 = final_w2[global_idx]
        if len(frame1.get('track_id', [])) == 0 or len(frame2.get('track_id', [])) == 0:
            continue
        pids1 = frame1['track_id'].numpy().astype(int) if isinstance(frame1['track_id'], torch.Tensor) else np.array(frame1['track_id']).astype(int)
        pids2 = frame2['track_id'].numpy().astype(int) if isinstance(frame2['track_id'], torch.Tensor) else np.array(frame2['track_id']).astype(int)
        for pid in sorted(set(pids1) & set(pids2)):
            if pid not in shapes:
                i1 = int(np.where(pids1 == pid)[0][0])
                i2 = int(np.where(pids2 == pid)[0][0])
                shapes[int(pid)] = {
                    'cam1': _to_numpy_shape(frame1['shape'], i1),
                    'cam2': _to_numpy_shape(frame2['shape'], i2),
                }
    return shapes


def compute_per_pid_scale_factors(per_pid_shapes, final_w1, final_w2, smplx_model, device):
    """Compute the scale factor (world metres / model units) per PID per camera.

    World height  : median Z-extent of joints_world across frames (Z is up in court coords).
    Model height  : Y-extent of the SMPLX T-pose vertices computed from the PID's own betas
                    (Y is up in SMPLX canonical space).

    Using T-pose vertices from actual betas avoids the 'joints' field, which may have a
    different row count than 'track_id' in interpolated frames produced by
    combine_world4ds_with_time_offset (joints_world IS properly handled; joints is not).

    Returns:
        dict {pid: {'cam1': float|None, 'cam2': float|None}}
    """
    from collections import defaultdict

    world_heights = [defaultdict(list), defaultdict(list)]

    for cam_idx, world4d in enumerate([final_w1, final_w2]):
        for frame_data in world4d.values():
            if len(frame_data.get('track_id', [])) == 0:
                continue
            if 'joints_world' not in frame_data:
                continue
            pids = (frame_data['track_id'].numpy().astype(int)
                    if isinstance(frame_data['track_id'], torch.Tensor)
                    else np.array(frame_data['track_id']).astype(int))
            joints_w = frame_data['joints_world']
            if hasattr(joints_w, 'cpu'):
                joints_w = joints_w.detach().cpu().numpy()
            for i, pid in enumerate(pids):
                if i >= len(joints_w):
                    continue
                j_w = joints_w[i]  # (25, 3), Z-up in court coords
                world_height = float(np.nanmax(j_w[:, 2]) - np.nanmin(j_w[:, 2]))
                if world_height > 0.1:
                    world_heights[cam_idx][int(pid)].append(world_height)

    result = {}
    for pid, shapes in per_pid_shapes.items():
        # Model height: Y-extent of T-pose verts from actual per-camera betas
        verts1 = compute_smplx_zero_pose_verts(shapes['cam1'], smplx_model, device)
        verts2 = compute_smplx_zero_pose_verts(shapes['cam2'], smplx_model, device)
        model_h1 = float(verts1[:, 1].max() - verts1[:, 1].min())
        model_h2 = float(verts2[:, 1].max() - verts2[:, 1].min())

        wh1 = float(np.median(world_heights[0][pid])) if world_heights[0][pid] else None
        wh2 = float(np.median(world_heights[1][pid])) if world_heights[1][pid] else None

        s1 = (wh1 / model_h1) if (wh1 is not None and model_h1 > 1e-6) else None
        s2 = (wh2 / model_h2) if (wh2 is not None and model_h2 > 1e-6) else None
        result[int(pid)] = {'cam1': s1, 'cam2': s2}
    return result


def compute_smplx_zero_pose_verts(betas, smplx_model, device):
    """Run SMPLX forward pass with zero pose (T-pose) and given betas.

    Returns vertices as np.ndarray of shape (V, 3).
    """
    zero_aa = torch.zeros(1, 55, 3, device=device)
    rotmat = axis_angle_to_matrix(zero_aa)  # (1, 55, 3, 3)
    betas_t = torch.tensor(betas, dtype=torch.float32, device=device).reshape(1, -1)
    transl = torch.zeros(1, 3, device=device)
    with torch.no_grad():
        output = smplx_model(
            global_orient=rotmat[:, :1],
            body_pose=rotmat[:, 1:22],
            betas=betas_t,
            transl=transl,
        )
    return output.vertices[0].cpu().numpy()  # (V, 3)


def _render_shape_panels(verts1, verts2, per_vertex_l2, cam1_name, cam2_name,
                          horiz_axis, vert_axis, horiz_label, vert_label, colorbar_label,
                          shift_to_origin=False):
    """Create a 2-row shape comparison figure and return it.

    Row 0: cam1 front | cam2 front | overlay (front) | per-vertex error (front)
    Row 1: cam1 side  | cam2 side  | per-vertex error (side)
    """
    depth_axis = next(a for a in range(3) if a not in (horiz_axis, vert_axis))
    unit = horiz_label.split('(')[-1].rstrip(')')
    depth_label_str = {0: 'x', 1: 'y', 2: 'z'}[depth_axis]
    side_horiz_label = f'{depth_label_str} ({unit})'

    if shift_to_origin:
        # Shift both meshes consistently so (0,0) is the bottom-left corner.
        verts1 = verts1.copy()
        verts2 = verts2.copy()
        for ax_idx in (horiz_axis, vert_axis, depth_axis):
            ax_min = min(verts1[:, ax_idx].min(), verts2[:, ax_idx].min())
            verts1[:, ax_idx] -= ax_min
            verts2[:, ax_idx] -= ax_min

    vmax = float(np.percentile(per_vertex_l2, 95)) if per_vertex_l2.max() > 0 else 1e-6
    dot_size = 0.5
    # Axis limits derived from actual vertex extents (wingspan / depth) with 5% padding
    x_max_front = max(verts1[:, horiz_axis].max(), verts2[:, horiz_axis].max()) * 1.05
    x_max_side  = max(verts1[:, depth_axis].max(),  verts2[:, depth_axis].max())  * 1.05

    mosaic = [
        ['c1f', 'c2f', 'ovf', 'errf'],
        ['c1s', 'c2s', 'errs',  '.'  ],
    ]
    fig, axd = plt.subplot_mosaic(mosaic, figsize=(24, 12))

    _FS_TITLE = 14
    _FS_LABEL = 13
    _FS_TICK  = 12

    def _style(a, title, xlabel, ylabel):
        a.set_title(title, fontsize=_FS_TITLE)
        a.set_xlabel(xlabel, fontsize=_FS_LABEL)
        a.set_ylabel(ylabel, fontsize=_FS_LABEL)
        a.tick_params(labelsize=_FS_TICK)
        a.set_aspect('equal')
        a.grid(True, alpha=0.2)

    # --- Row 0: front views ---
    ax = axd['c1f']
    ax.scatter(verts1[:, horiz_axis], verts1[:, vert_axis], c='steelblue', s=dot_size, alpha=0.6)
    _style(ax, cam1_name, horiz_label, vert_label)
    ax.set_xlim(0, x_max_front); ax.set_ylim(0, 2)

    ax = axd['c2f']
    ax.scatter(verts2[:, horiz_axis], verts2[:, vert_axis], c='darkorange', s=dot_size, alpha=0.6)
    _style(ax, cam2_name, horiz_label, vert_label)
    ax.set_xlim(0, x_max_front); ax.set_ylim(0, 2)

    ax = axd['ovf']
    ax.scatter(verts1[:, horiz_axis], verts1[:, vert_axis],
               c='steelblue', s=dot_size, alpha=0.5)
    ax.scatter(verts2[:, horiz_axis], verts2[:, vert_axis],
               c='darkorange', s=dot_size, alpha=0.5)
    _style(ax, 'Overlay (front)', horiz_label, vert_label)
    ax.set_xlim(0, x_max_front); ax.set_ylim(0, 2)

    ax = axd['errf']
    sc = ax.scatter(verts1[:, horiz_axis], verts1[:, vert_axis],
                    c=per_vertex_l2, cmap='hot', s=dot_size, vmin=0, vmax=vmax, alpha=0.8)
    plt.colorbar(sc, ax=ax, label=colorbar_label, fraction=0.046, pad=0.04).ax.tick_params(labelsize=_FS_TICK)
    _style(ax, 'Per-vertex error (front)', horiz_label, vert_label)
    ax.set_xlim(0, x_max_front); ax.set_ylim(0, 2)

    # --- Row 1: side views ---
    ax = axd['c1s']
    ax.scatter(verts1[:, depth_axis], verts1[:, vert_axis], c='steelblue', s=dot_size, alpha=0.6)
    _style(ax, f'{cam1_name} (side)', side_horiz_label, vert_label)
    ax.set_xlim(0, x_max_side); ax.set_ylim(0, 2)

    ax = axd['c2s']
    ax.scatter(verts2[:, depth_axis], verts2[:, vert_axis], c='darkorange', s=dot_size, alpha=0.6)
    _style(ax, f'{cam2_name} (side)', side_horiz_label, vert_label)
    ax.set_xlim(0, x_max_side); ax.set_ylim(0, 2)

    ax = axd['errs']
    sc2 = ax.scatter(verts1[:, depth_axis], verts1[:, vert_axis],
                     c=per_vertex_l2, cmap='hot', s=dot_size, vmin=0, vmax=vmax, alpha=0.8)
    plt.colorbar(sc2, ax=ax, label=colorbar_label, fraction=0.046, pad=0.04).ax.tick_params(labelsize=_FS_TICK)
    _style(ax, 'Per-vertex error (side)', side_horiz_label, vert_label)
    ax.set_xlim(0, x_max_side); ax.set_ylim(0, 2)

    return fig

def save_pid_rgb_crops(per_pid_shapes, final_w1, final_w2, output_dir,
                       video_base_path_1, video_base_path_2):
    """Save a cropped RGB image for each PID so it's easy to identify who is who
    in the SMPLX visualizations.

    For each PID, searches final_w1 then final_w2 for the first frame where the
    person is visible with a valid video frame. Projects the person's world-space
    joints into the image, crops around them, and saves as
    ``smplx_pid{pid}_rgb_crop.png`` in output_dir.
    """
    import cv2

    os.makedirs(output_dir, exist_ok=True)

    for pid in sorted(per_pid_shapes.keys()):
        saved = False
        for world4d, vbp in [(final_w1, video_base_path_1),
                             (final_w2, video_base_path_2)]:
            for frame_idx in sorted(world4d.keys()):
                fd = world4d[frame_idx]
                og_frame = fd.get('og_rgb_frame')
                og_vpath = fd.get('og_video_path')
                if og_frame is None or og_vpath is None:
                    continue

                track_ids = fd.get('track_id', [])
                if hasattr(track_ids, 'numpy'):
                    track_ids = track_ids.numpy().astype(int).tolist()
                else:
                    track_ids = [int(t) for t in track_ids]
                if pid not in track_ids:
                    continue

                frame_bgr = extract_frame_from_video(og_vpath, og_frame, vbp)
                if frame_bgr is None:
                    continue

                frame_rgb = cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB)
                person_idx = track_ids.index(pid)
                joints = fd['joints_world'][person_idx]
                if hasattr(joints, 'numpy'):
                    joints = joints.numpy()

                cam = fd.get('camera')
                K = fd.get('K_calib')
                if cam is not None and K is not None:
                    R_c2w = cam[:3, :3]
                    T_c2w = cam[:3, 3]
                    R_w2c = R_c2w.T
                    t_w2c = -R_w2c @ T_c2w
                    pts_2d = []
                    for joint in joints:
                        P_cam = R_w2c @ joint + t_w2c
                        if P_cam[2] <= 0:
                            continue
                        uvw = K @ P_cam
                        pts_2d.append((uvw[0] / uvw[2], uvw[1] / uvw[2]))

                    if pts_2d:
                        H, W = frame_rgb.shape[:2]
                        us = [p[0] for p in pts_2d]
                        vs = [p[1] for p in pts_2d]
                        bw = max(us) - min(us)
                        bh = max(vs) - min(vs)
                        pad = max(bw, bh) * 0.4
                        x1 = max(0, int(min(us) - pad))
                        y1 = max(0, int(min(vs) - pad))
                        x2 = min(W, int(max(us) + pad))
                        y2 = min(H, int(max(vs) + pad))
                        if x2 > x1 and y2 > y1:
                            frame_rgb = frame_rgb[y1:y2, x1:x2]

                out_path = os.path.join(output_dir, f'smplx_pid{pid}_rgb_crop.png')
                plt.imsave(out_path, frame_rgb)
                print(f"  Saved RGB crop: {out_path}")
                saved = True
                break  # Found a good frame for this camera

            if saved:
                break  # Move on to the next PID

        if not saved:
            print(f"  Warning: no valid RGB frame found for PID {pid}")


def plot_smplx_shape_comparison(per_pid_shapes, smplx_model, cam1_name, cam2_name,
                                 device, output_dir, scale_by_pid=None):
    """Render SMPLX T-pose for each PID from each camera, showing per-vertex L2 error.

    Produces two figures per PID:
      Figure 1 (model units): vertices in the SMPLX canonical coordinate system.
          Front view x vs y, side view z vs y, all axes in 'model units'.
      Figure 2 (world scale): vertices scaled to metric height using
          scale_by_pid[pid] derived from world-space vs model-space joint extents.
          Shown only when scale_by_pid is provided.

    Also saves each figure as a PNG in output_dir.

    Returns:
        figs: list of (name, figure)
        vertex_errors: {pid: {'per_vertex_l2_model': list, 'mean_l2_model': float,
                               'per_vertex_l2_world': list, 'mean_l2_world': float,  (if scaled)
                               'scale_cam1': float, 'scale_cam2': float}}
    """
    figs = []
    vertex_errors = {}

    for pid, shapes in sorted(per_pid_shapes.items()):
        betas1 = shapes['cam1']
        betas2 = shapes['cam2']

        verts1 = compute_smplx_zero_pose_verts(betas1, smplx_model, device)  # (V,3) model units
        verts2 = compute_smplx_zero_pose_verts(betas2, smplx_model, device)  # (V,3) model units

        per_vertex_l2_model = np.linalg.norm(verts1 - verts2, axis=1)
        beta_l2 = float(np.linalg.norm(betas1 - betas2))

        pid_scales = (scale_by_pid or {}).get(pid, {})
        s1 = pid_scales.get('cam1', None)
        s2 = pid_scales.get('cam2', None)

        # ---- Figure 1: model units ----
        fig1 = _render_shape_panels(
            verts1, verts2, per_vertex_l2_model,
            cam1_name, cam2_name,
            horiz_axis=0, vert_axis=1,
            horiz_label='x (model units)', vert_label='y (model units)',
            colorbar_label='L2 (model units)',
            shift_to_origin=True,
        )
        fig1.suptitle(
            f'SMPLX T-Pose Shape Comparison [model units]  |  PID {pid}  |  '
            f'{cam1_name} vs {cam2_name}\n'
            f'Beta L2: {beta_l2:.4f}   '
            f'Mean vertex L2: {float(np.mean(per_vertex_l2_model)):.4f}   '
            f'Max vertex L2: {float(np.max(per_vertex_l2_model)):.4f}',
            fontsize=15,
        )
        fig1.tight_layout()
        if output_dir:
            p = os.path.join(output_dir, f'smplx_shape_pid{pid}_model_units.png')
            fig1.savefig(p, dpi=120, bbox_inches='tight')
            print(f"  Saved: {p}")
        figs.append((f'SMPLX Shape PID {pid} (model units)', fig1))

        # Record model-unit errors
        entry = {
            'per_vertex_l2_model': per_vertex_l2_model.tolist(),
            'mean_l2_model': float(np.mean(per_vertex_l2_model)),
            'max_l2_model': float(np.max(per_vertex_l2_model)),
            'scale_cam1': s1,
            'scale_cam2': s2,
        }

        # ---- Figure 2: world scale (only when scale is available) ----
        if s1 is not None and s2 is not None:
            verts1_w = verts1 * s1   # (V,3) metres
            verts2_w = verts2 * s2   # (V,3) metres
            per_vertex_l2_world = np.linalg.norm(verts1_w - verts2_w, axis=1)

            fig2 = _render_shape_panels(
                verts1_w, verts2_w, per_vertex_l2_world,

def plot_beta_consistency(beta_l2_by_pid, cam1_name, cam2_name):

def plot_shape_chunk_consistency(chunk_shapes_1, chunk_shapes_2, cam1_name, cam2_name):
    """Two figures for shape consistency across PromptHMR chunks.

    Figure 1: L2 drift vs chunk index (one line per PID, one subplot per camera).
    Figure 2: 2-D scatter of beta[0] vs beta[1], colored by PID,
              markers differentiated by camera.

    Returns list of (name, figure) tuples.
    """
    COLORS = ['tab:blue', 'tab:orange', 'tab:green', 'tab:red', 'tab:purple',
              'tab:brown', 'tab:pink', 'tab:gray']

    all_pids = sorted(set(chunk_shapes_1.keys()) | set(chunk_shapes_2.keys()))
    pid_color = {pid: COLORS[i % len(COLORS)] for i, pid in enumerate(all_pids)}

    # ---- Figure 1: L2 drift vs chunk start frame ----
    fig1, axes = plt.subplots(1, 2, figsize=(14, 5), sharey=True)
    for ax, cam_shapes, cam_name in [(axes[0], chunk_shapes_1, cam1_name),
                                      (axes[1], chunk_shapes_2, cam2_name)]:
        for pid, chunk_dict in cam_shapes.items():
            chunk_starts = sorted(chunk_dict.keys())
            if len(chunk_starts) < 2:
                continue
            ref_shape = chunk_dict[chunk_starts[0]]
            l2s = [float(np.linalg.norm(chunk_dict[cs] - ref_shape)) for cs in chunk_starts]
            ax.plot(chunk_starts, l2s, marker='o',
                    color=pid_color[pid], label=f'PID {pid}')
        ax.set_title(f'{cam_name}')
        ax.set_xlabel('Chunk start frame')
        ax.set_ylabel('Beta L2 vs first chunk')
        ax.legend(fontsize=8)
        ax.grid(True, alpha=0.3)
    fig1.suptitle('Shape (Beta) L2 Drift Across PromptHMR Chunks', fontsize=13)
    fig1.subplots_adjust(top=0.88, bottom=0.12, left=0.08, right=0.97, wspace=0.15)

    # ---- Figure 2: beta[0] vs beta[1] scatter ----
    fig2, ax2 = plt.subplots(figsize=(8, 6))
    for pid in all_pids:
        color = pid_color[pid]
        if pid in chunk_shapes_1:
            pts = np.array(list(chunk_shapes_1[pid].values()))  # (K, 10)
            ax2.scatter(pts[:, 0], pts[:, 1], color=color, marker='o', s=60,
                        label=f'PID {pid} ({cam1_name})', alpha=0.8)
        if pid in chunk_shapes_2:
            pts = np.array(list(chunk_shapes_2[pid].values()))  # (K, 10)
            ax2.scatter(pts[:, 0], pts[:, 1], color=color, marker='^', s=60,
                        label=f'PID {pid} ({cam2_name})', alpha=0.8, edgecolors='black',
                        linewidths=0.5)
    ax2.set_xlabel('Beta[0]')
    ax2.set_ylabel('Beta[1]')
    ax2.set_title('Shape Parameter Scatter Across Chunks\n(circles = cam1, triangles = cam2)')
    ax2.legend(fontsize=8)
    ax2.grid(True, alpha=0.3)
    fig2.subplots_adjust(top=0.88, bottom=0.12, left=0.10, right=0.97)

    return [("Shape Chunk L2 Drift", fig1), ("Shape Beta Scatter", fig2)]


def plot_chunk_degradation(degradation_data):
    """Plot average pairwise error vs time-within-chunk with ±1 std band."""
    fig, ax = plt.subplots(figsize=(10, 5))
    if not degradation_data:
        ax.text(0.5, 0.5, 'No data available', ha='center', va='center',
                transform=ax.transAxes)
        ax.set_title('Cross-Camera Error vs Position Within PromptHMR Chunk')
        fig.tight_layout()
        return fig

    t = degradation_data['t_sec']
    metrics = [
        ('trans_l2_mean', 'trans_l2_std', 'Translation L2', 'blue'),
        ('pose_l2_mean',  'pose_l2_std',  'Procrustes Pose L2', 'orange'),
        ('mpjpe_mean',    'mpjpe_std',    'MPJPE', 'red'),
    ]
    for mean_key, std_key, label, color in metrics:
        if mean_key not in degradation_data:
            continue
        mean = degradation_data[mean_key]
        std = degradation_data[std_key]
        ax.plot(t, mean, label=label, color=color, linewidth=1.5)
        ax.fill_between(t, mean - std, mean + std, color=color, alpha=0.15)

    ax.set_xlabel('Time within chunk (s)')
    ax.set_ylabel('Pairwise L2 (m)')
    ax.set_title('Cross-Camera Error vs Position Within PromptHMR Chunk\n(shaded = ±1 std across chunks)')
    ax.legend()
    ax.grid(True, alpha=0.3)
    fig.tight_layout()
    return fig

                # Check if either camera has drift at this frame

                    if any(d == 1 for d in pose_drift_2):
                        has_drift = 1
                drift_flags.append(has_drift)
            return np.array(drift_flags)

        fig = visualize_frame_comparison_grid_court(final_w1, final_w2, best_frames_final,
                                                    cam1_name=cam1_name, cam2_name=cam2_name)
        all_figures.append(("Best Aligned Poses, Court View (After)", fig))
        fig = visualize_frame_comparison_grid_zoomed(final_w1, final_w2, best_frames_final,
                                                        cam1_name=cam1_name, cam2_name=cam2_name,
                                                        title="Best Aligned Poses, Zoomed View (After Calibration)")

        # Worst aligned poses - Court view
        fig = visualize_frame_comparison_grid_court(final_w1, final_w2, worst_frames_final,
                                                        cam1_name=cam1_name, cam2_name=cam2_name,
                                                        title="Worst Aligned Poses, Court View (After Calibration)")
        all_figures.append(("Worst Aligned Poses, Court View (After)", fig))
        fig = visualize_frame_comparison_grid_zoomed(final_w1, final_w2, worst_frames_final,
                                                        cam1_name=cam1_name, cam2_name=cam2_name)
        all_figures.append(("Worst Aligned Poses, Zoomed View (After)", fig))

        # --- SAVE RESULTS ---
        print("\n[Saving Results]...")
        
        results = {
            'chunk_starts': chunk_starts,
            'all_chunk_starts': all_chunk_starts,
            'coarse_offsets': coarse_best,
            'fine_offsets': fine_best,
            'median_offset': median_offset,
            'calibrate_frame_subset': args.calibrate_frame_subset,

            'coarse_search_params': {
                'start_offset': args.start_offset,

                'offset_step': args.offset_step,
            },
            'fine_search_params': {
                'range': fine_range,
                'step': fine_step,

            'cam1_name': cam1_name,
            'cam2_name': cam2_name,

        results_path = os.path.join(args.output_dir, "calibration_results.pkl")
        joblib.dump(results, results_path)
        print(f"  Results saved to: {results_path}")

        # Save aligned world4d dictionaries NOW (intersection only, before temporal union)
        # so the pkl files stay small and fast to load on subsequent runs.
        joblib.dump(final_w1, w1_aligned_path)
        joblib.dump(final_w2, w2_aligned_path)
        open(_pid_matched_flag, 'w').close()
        print(f"  Aligned world4d 1 saved to: {w1_aligned_path}")
        print(f"  Aligned world4d 2 saved to: {w2_aligned_path}")

        # Extend final_w1/final_w2 (in memory only) to the full temporal union so
        # the fused output covers the full video duration.  The saved pkl files above
        # remain intersection-only for fast loading on subsequent runs.
        print("\n[Extending to temporal union (in-memory only)]...")
        ext_idx = global_idx  # continue sequential indexing

        # Helper to build an empty frame dict for one camera in the other's slot
        def _make_ext_empty(ref_frame):
            return {
                'track_id': np.array([]),
                'camera': ref_frame.get('camera', np.eye(4)),
                'global_timestamp': ref_frame.get('global_timestamp'),
                'global_time_start': ref_frame.get('global_time_start'),
                'relative_timestamp': ref_frame.get('relative_timestamp'),
                'og_rgb_frame': None,
                'og_video_path': None,
            }

        # Camera 1 frames outside the common range (camera 2 gets empty for those indices)
        cam1_extra = sorted(
            [k for k, v in w4d1_full.items()
             if v.get('global_timestamp') is not None
             and (v['global_timestamp'] < start_common or v['global_timestamp'] > end_common)],
            key=lambda k: w4d1_full[k]['global_timestamp']
        )
        for k in cam1_extra:
            final_w1[ext_idx] = w4d1_full[k]
            final_w2[ext_idx] = _make_ext_empty(w4d1_full[k])
            ext_idx += 1

        # Camera 2 frames outside the common range — apply temporal offset so timestamps
        # are expressed in camera 1's time domain (times_2_shifted = times_2 + time_offset).
        cam2_extra = sorted(
            [k for k, v in w4d2_full.items()
             if v.get('global_timestamp') is not None
             and (v['global_timestamp'] + median_offset < start_common
                  or v['global_timestamp'] + median_offset > end_common)],
            key=lambda k: w4d2_full[k]['global_timestamp']
        )
        for k in cam2_extra:
            frame2 = dict(w4d2_full[k])
            # Shift the timestamp into camera 1's domain
            if frame2.get('global_timestamp') is not None:
                frame2['global_timestamp'] = frame2['global_timestamp'] + median_offset
            final_w1[ext_idx] = _make_ext_empty(frame2)
            final_w2[ext_idx] = frame2
            ext_idx += 1

        print(f"  Added {len(cam1_extra)} extra camera-1 frames and "
              f"{len(cam2_extra)} extra camera-2 frames (temporal union)")
        print(f"  final_w1: {len(final_w1)} frames, final_w2: {len(final_w2)} frames")

        # Note: PDF generation moved to end of script to include post-processing figures

        # Print summary
        print("\n" + "="*60)
        print("CALIBRATION COMPLETE")
        print("="*60)

        valid_coarse = [x for x in coarse_best if x is not None]
        valid_fine = [x for x in fine_best if x is not None]

        if valid_coarse:
            print(f"  Coarse offsets: mean={np.mean(valid_coarse):.1f}ms, std={np.std(valid_coarse):.1f}ms")
        if valid_fine:
            print(f"  Fine offsets: mean={np.mean(valid_fine):.1f}ms, std={np.std(valid_fine):.1f}ms")

        print(f"\n  Results: {args.output_dir}")

    # =========================================================
    # POST-CALIBRATION: FUSION AND VIDEO GENERATION
    # =========================================================
    # This section runs whether calibration was just performed or loading existing results

    print("\n" + "="*60)
    print("POST-PROCESSING: FUSION AND VIDEO GENERATION")
    print("="*60)

    # Collect post-processing figures for PDF
    post_figures = []

    # Load the aligned world4d files
    print("\n[Loading Aligned World4Ds]...")
    final_w1 = joblib.load(w1_aligned_path)
    final_w2 = joblib.load(w2_aligned_path)
    print(f"  Loaded {len(final_w1)} frames from camera 1")
    print(f"  Loaded {len(final_w2)} frames from camera 2")

    # Refresh og_video_path for the current machine (paths stored in the pkl may be

    # Only og_video_path is updated; og_rgb_frame is left intact.
    def _refresh_video_paths(world4d, video_base_path, world4d_path):
        import glob as _glob
        if not video_base_path or not world4d_path:
            return
        session_name = os.path.basename(os.path.dirname(world4d_path))
        new_path = None
        for ext in ['.MOV', '.mov', '.mp4', '.MP4']:
            matches = _glob.glob(os.path.join(video_base_path, '*', session_name + ext))
            if matches:
                new_path = matches[0][:-len(ext)]
                print(f"  [refresh_paths] Found video: {new_path}{ext}")
                break
        if new_path is None:
            new_path = video_base_path  # fallback
        for frame_data in world4d.values():
            frame_data['og_video_path'] = new_path

    _refresh_video_paths(final_w1, args.video_base_path_1, args.world4d_path_1)
    _refresh_video_paths(final_w2, args.video_base_path_2, args.world4d_path_2)

    # --- CROP FRAMES FOR PROCESSING ---
    # Crop to max_num_frames if specified (for faster testing)
    if args.max_num_frames is not None:
        keys1 = sorted(final_w1.keys())
        keys2 = sorted(final_w2.keys())
        keep1 = keys1[:args.max_num_frames]
        keep2 = keys2[:args.max_num_frames]
        final_w1 = {k: final_w1[k] for k in keep1}
        final_w2 = {k: final_w2[k] for k in keep2}
        print(f"  Cropped to --max_num_frames={args.max_num_frames}:")
        print(f"    final_w1: {len(keys1)} -> {len(final_w1)} frames")
        print(f"    final_w2: {len(keys2)} -> {len(final_w2)} frames")

    # --- POSE FRAGMENTATION METRICS (recompute if calibration was skipped) ---
    # When calibration was already done (aligned files existed), fragmentation was
    # not computed in the else branch above. Recompute here on the aligned world4ds.
    if w4d1 is None or naive_switch_w1['count'] == 0 and not cvtss_dict['confirmed_events']:
        print("\n[Computing Pose Fragmentation Metrics]...")
        naive_switch_w1 = compute_naive_switch_count(final_w1, spatial_threshold=0.2)
        naive_switch_w2 = compute_naive_switch_count(final_w2, spatial_threshold=0.2)
        cvtss_dict = compute_cvtss(
            [final_w1, final_w2], max_gap_frames=30, spatial_threshold=1.0, confirmation_radius=1.0
        )
        cvtss_str_pp = f"{cvtss_dict['cvtss']:.3f}" if not (
            isinstance(cvtss_dict['cvtss'], float) and np.isnan(cvtss_dict['cvtss'])) else "N/A"
        print(f"  {cam1_name}: {naive_switch_w1['count']} naive switches")
        print(f"  {cam2_name}: {naive_switch_w2['count']} naive switches")
        print(f"  CVTSS: {cvtss_str_pp} ({cvtss_dict['num_confirmed']}/{cvtss_dict['num_candidates']})")
        w4d1 = final_w1
        w4d2 = final_w2

    # --- SPATIAL FUSION ---
    fused_path = os.path.join(args.output_dir, "world4d_fused.pkl")

    if os.path.exists(fused_path) and not args.force:
        print("\n[Spatial Fusion] Loading existing fused world4d...")
        world4d_fused = joblib.load(fused_path)
        # pdb.set_trace()
        print(f"  Loaded {len(world4d_fused)} frames from: {fused_path}")
    else:
        print("\n[Spatial Fusion]...")
        world4d_fused = spatial_fusion_mle(final_w1, final_w2)
        print(f"  Fused {len(world4d_fused)} frames")

        # Save fused world4d
        joblib.dump(world4d_fused, fused_path)
        print(f"  Fused world4d saved to: {fused_path}")

    # --- IDENTIFY POSE DRIFT ---
    print("\n[Identifying Pose Drift]...")
    final_w1, drift_stats_1 = identify_pose_drift(final_w1, verbose=False)
    final_w2, drift_stats_2 = identify_pose_drift(final_w2, verbose=False)
    _, drift_stats_fused = identify_pose_drift(world4d_fused, verbose=False)
    print(f"  Camera 1: {drift_stats_1['drift_poses']}/{drift_stats_1['total_poses']} poses drifting "
          f"({drift_stats_1['drift_ratio']*100:.1f}%)")
    print(f"  Camera 2: {drift_stats_2['drift_poses']}/{drift_stats_2['total_poses']} poses drifting "
          f"({drift_stats_2['drift_ratio']*100:.1f}%)")

    # Find drift segments for potential video generation
    drift_segments_1 = get_drift_segments(final_w1, min_segment_length=10)
    drift_segments_2 = get_drift_segments(final_w2, min_segment_length=10)
    # Drift speed plot
    fig_drift_speed = plot_pose_drift_speed(
        drift_stats_1, drift_stats_2, drift_stats_fused,
        cam1_name=cam1_name, cam2_name=cam2_name,
        title="Drift Detection: Pose Translation Speed vs Frame"
    )
    if fig_drift_speed:
        post_figures.append(("Pose Drift Speed", fig_drift_speed))

    # --- Compute errors from fused world4d (needed for statistics and best/worst frame selection) ---
    print("\n[Computing pose alignment errors from fused data]...")
    err_fused = compute_fused_errors(world4d_fused)
    error_analysis_dict["errors_fused"] = err_fused

    # --- Feature 1: Cross-camera beta (shape) consistency + SMPLX T-pose rendering ---
    print("\n[Computing cross-camera beta consistency]...")
    beta_l2_by_pid = compute_beta_consistency(final_w1, final_w2)
    error_analysis_dict['shape_beta_consistency'] = beta_l2_by_pid
    print(f"  Beta L2 per PID: {beta_l2_by_pid}")
    fig_beta = plot_beta_consistency(beta_l2_by_pid, cam1_name, cam2_name)
    post_figures.append(("Beta (Shape) Consistency", fig_beta))

    # SMPLX T-pose render + per-vertex error
    print("\n[Computing SMPLX T-pose shape renders and per-vertex error]...")
    try:
        _smplx_device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
        _smplx_model = SMPLX_Layer(SMPLX_PATH).to(_smplx_device)
        per_pid_shapes = extract_per_pid_shapes(final_w1, final_w2)
        if per_pid_shapes:
            scale_by_pid = compute_per_pid_scale_factors(
                per_pid_shapes, final_w1, final_w2, _smplx_model, _smplx_device)
            print(f"  Scale factors (world m / model unit): "
                  f"{ {pid: {k: round(v, 3) for k, v in d.items() if v is not None} for pid, d in scale_by_pid.items()} }")
            smplx_figs, smplx_vertex_errors = plot_smplx_shape_comparison(
                per_pid_shapes, _smplx_model, cam1_name, cam2_name,

                post_figures.append((name, fig))
            error_analysis_dict['shape_vertex_consistency'] = smplx_vertex_errors
            print(f"  Mean vertex L2 (model units): "
                  f"{ {pid: round(v['mean_l2_model'], 5) for pid, v in smplx_vertex_errors.items()} }")
            print("\n[Saving PID RGB crops for SMPLX identification]...")
            save_pid_rgb_crops(
                per_pid_shapes, final_w1, final_w2,
                args.output_dir,
                args.video_base_path_1, args.video_base_path_2,
            )
        else:
            print("  No matched PIDs found; skipping SMPLX render.")
        del _smplx_model
    except Exception as e:
        print(f"  Warning: SMPLX T-pose render failed: {e}")

    # --- Feature 2: Shape consistency across PromptHMR chunks ---
    if args.phmr_chunk_size > 0:
        print("\n[Computing shape consistency across PromptHMR chunks]...")
        chunk_shapes_1 = collect_chunk_shapes(final_w1, args.phmr_chunk_size, args.phmr_chunk_step)
        chunk_shapes_2 = collect_chunk_shapes(final_w2, args.phmr_chunk_size, args.phmr_chunk_step)
        print(f"  Camera 1: {len(chunk_shapes_1)} PIDs across chunks")
        print(f"  Camera 2: {len(chunk_shapes_2)} PIDs across chunks")
        figs_chunk_shape = plot_shape_chunk_consistency(chunk_shapes_1, chunk_shapes_2, cam1_name, cam2_name)

            post_figures.append((name, fig))

                        for pid, pdict in chunk_shapes_1.items()},
            cam2_name: {str(pid): {str(cs): s.tolist() for cs, s in pdict.items()}
                        for pid, pdict in chunk_shapes_2.items()},
        }

            final_w1, final_w2, args.fps, args.phmr_chunk_size, args.phmr_chunk_step)
        if degradation_data:
            print(f"  Collected data for {len(degradation_data['t_sec'])} positions within chunk")
            error_analysis_dict['temporal_chunk_degradation'] = {
                k: v.tolist() if isinstance(v, np.ndarray) else v

    print("\n[Analyzing Joint Error Correlations - Fused]...")

    post_figures.extend(corr_figs_fused)
    if fused_corr_matrix is not None:
        error_analysis_dict['joint_correlations'] = {
            'joint_names': fused_corr_joint_names,
            'correlation_matrix': fused_corr_matrix.tolist(),

    print("\n[Computing camera-vs-fused errors]...")
    error_dict = compute_camera_vs_fused_error_dict_two_cam(

    # Compute per-frame average depth (distance from person to each camera center).
    # camera_1 / camera_2 are 4x4 c2w matrices stored in every fused frame; the
    # camera center in world space is the last column [:3, 3].
    _avg_depth_per_frame = []
    for _fidx in sorted(world4d_fused.keys()):
        _frame = world4d_fused[_fidx]
        _tw = _frame.get('trans_world')
        if _tw is not None:
            try:
                _tw_np = np.asarray(_tw.numpy() if hasattr(_tw, 'numpy') else _tw, dtype=float)
                if _tw_np.ndim == 1:
                    _tw_np = _tw_np[np.newaxis, :]
                if _tw_np.shape[0] > 0:
                    _c1 = np.asarray(_frame.get('camera_1', np.eye(4)), dtype=float)[:3, 3]
                    _c2 = np.asarray(_frame.get('camera_2', np.eye(4)), dtype=float)[:3, 3]
                    _cam_centers = np.stack([_c1, _c2])  # (2, 3)
                    _depths = np.linalg.norm(
                        _tw_np[:, np.newaxis, :] - _cam_centers[np.newaxis, :, :], axis=-1
                    )  # (N_people, 2)
                    _avg_depth_per_frame.append(float(np.mean(_depths)))
                    continue
            except Exception:
                pass
        _avg_depth_per_frame.append(None)

    error_dict['mean_fused_pose_error']['avg_depth_m'] = _avg_depth_per_frame
    error_analysis_dict["camera_vs_fused_errors"] = error_dict

    # Plot error dict timeseries for each metric
    groups = [
        [('trans_l2', 'Translation L2 (m)'), ('pose_l2', 'Pose L2 (m)')],
        [('mpjpe', 'MPJPE (m)'), ('pa_mpjpe', 'PA-MPJPE (m)')]
    ]

    for group in groups:
        # Create a page with 2 subplots
        fig, axes = plt.subplots(2, 1, figsize=FIGURE_SIZE, sharex=True)
        
        # Fill the subplots using your helper
        plot_error_dict_timeseries(error_dict, group[0][0], f"Camera vs Fused: {group[0][1]}", 
                                   ylabel=group[0][1], cam_names=[cam1_name, cam2_name], ax=axes[0])
        
        plot_error_dict_timeseries(error_dict, group[1][0], f"Camera vs Fused: {group[1][1]}", 
                                   ylabel=group[1][1], cam_names=[cam1_name, cam2_name], ax=axes[1])
        
        # Final housekeeping for the page
        axes[1].set_xlabel('Timestamp / Frame Index', fontsize=12)
        plt.tight_layout()
        
        # Append to your figure list
        page_label = f"Fused Comparison {group[0][0]}_{group[1][0]}"
        post_figures.append((page_label, fig))

    # Find best and worst pose indices
    all_aligned_keys = sorted(final_w1.keys())
    valid_mask = err_fused['has_match'] & ~np.isnan(err_fused['pose_l2_procrustes_mean'])
    valid_indices = err_fused['frame_idx'][valid_mask]
    valid_errors = err_fused['pose_l2_procrustes_mean'][valid_mask]

    if len(valid_indices) > 0:

        best_pose_idx, worst_pose_idx = get_best_worst_frames(err_fused, n_best=3, n_worst=3)
        # valid_indices/valid_errors are positional; best/worst_pose_idx are frame keys — build a lookup
        frame_error_map = dict(zip(valid_indices.tolist(), valid_errors.tolist()))
        print(f"  Best pose alignment at frame {best_pose_idx} (error: {frame_error_map.get(best_pose_idx[0], float('nan')):.4f}m)")
        print(f"  Worst pose alignment at frame {worst_pose_idx} (error: {frame_error_map.get(worst_pose_idx[0], float('nan')):.4f}m)")
    else:
        best_pose_idx = [all_aligned_keys[0]]
        worst_pose_idx = [all_aligned_keys[-1]]
        print("  Warning: No valid frames for error computation, using first/last frames")

    # --- FOOT SKATING METRIC ---
    print("\n[Computing Foot Skating Metric]...")

    skating_w1 = compute_foot_skating_metric(final_w1, fps=args.fps)
    skating_w2 = compute_foot_skating_metric(final_w2, fps=args.fps)
    skating_fused = compute_foot_skating_metric(world4d_fused, fps=args.fps)
    error_analysis_dict["physics_foot_skating"] = {
        "cam1": skating_w1,
        "cam2": skating_w2,
        "fused": skating_fused
    }

    # feet plots
    skating_metrics = [skating_w1, skating_w2, skating_fused]
    skating_labels = [cam1_name, cam2_name, "Fused Pose"]
    fig = plot_foot_skating_comparison(skating_metrics, skating_labels)
    post_figures.append(("Foot Skating", fig))

    # --- STABILITY METRIC ---
    print("\n[Computing Stability Metric]...")
    stability_w1 = compute_stability_metric(final_w1)
    stability_w2 = compute_stability_metric(final_w2)
    stability_fused = compute_stability_metric(world4d_fused)
    error_analysis_dict["physics_stability"] = {
        "cam1": stability_w1,
        "cam2": stability_w2,
        "fused": stability_fused
    }

    stability_metrics = [stability_w1, stability_w2, stability_fused]
    stability_labels = [cam1_name, cam2_name, "Fused Pose"]
    fig = plot_stability_unified(stability_metrics, stability_labels)
    post_figures.append(("Unified Stability Analysis", fig))

    # --- CROSS-VIEW CONSISTENCY METRICS ---
    print("\n[Computing Cross-View Consistency Metrics]...")
    consistency_metrics = compute_cross_view_consistency_metrics(final_w1, final_w2)
    print(f"  Stability consistency:    {consistency_metrics['stability_consistency_mean']:.4f}m")
    print(f"  Foot vel. consistency:    {consistency_metrics['foot_velocity_consistency_mean']:.4f}m/s")
    print(f"  Foot height consistency:  {consistency_metrics['foot_height_consistency_mean']:.4f}m")
    error_analysis_dict["cross_view_consistency"] = {
        "stability_consistency": round(float(consistency_metrics['stability_consistency_mean']), 6),
        "foot_velocity_consistency": round(float(consistency_metrics['foot_velocity_consistency_mean']), 6),
        "foot_velocity_left": round(float(consistency_metrics['foot_velocity_left_mean']), 6),
        "foot_velocity_right": round(float(consistency_metrics['foot_velocity_right_mean']), 6),
        "foot_height_consistency": round(float(consistency_metrics['foot_height_consistency_mean']), 6),

    # --- Timestamp Alignment Verification ---
    # Use video_base_path_1 as the lookup root; og_video_path in the frames is absolute

        [final_w1, final_w2],

    )

    generate_worst_frames_pdf(
        world4d_fused=world4d_fused,
        final_w1=final_w1,
        final_w2=final_w2,
        err_fused=err_fused,

        output_folder_name=output_folder_name,
    )

    # --- GENERATE FINAL PDF (combining calibration and post-processing figures) ---
    print("\n[Generating PDF Analysis]...")
    pdf_filename = f"{output_folder_name}.pdf"
    pdf_path = os.path.join(args.output_dir, pdf_filename)

    # Combine all figures (calibration + post-processing)
    combined_figures = []
    if 'all_figures' in dir() and all_figures:
        combined_figures.extend(all_figures)
    if post_figures:
        combined_figures.extend(post_figures)

    if combined_figures:
        generate_summary_pdf_vector(pdf_path, combined_figures)
        print(f"  PDF saved to: {pdf_path}")

                plt.close(fig)
            except Exception:
                pass
