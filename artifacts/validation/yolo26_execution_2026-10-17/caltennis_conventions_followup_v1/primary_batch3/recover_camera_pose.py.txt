def extract_intrinsics_from_exif(video_path, image_width=1920, image_height=1080):
    """
    Attempts to get focal length from EXIF. 
    Falls back to a heuristic if metadata is missing.
    """
    f_p = None

    try:
        # 1. Attempt ExifTool Extraction
        cmd = ["exiftool", "-FocalLengthIn35mmFormat", "-j", str(video_path)]
        result = subprocess.run(cmd, capture_output=True, text=True)
        metadata = json.loads(result.stdout)[0]
        
        f_35mm = metadata.get("FocalLengthIn35mmFormat")
        
        if f_35mm is not None:
            if isinstance(f_35mm, str):
                f_35mm = float(f_35mm.split()[0])
            
            # Convert 35mm equiv to pixels
            f_p = image_width * (f_35mm / 36.0)

    except Exception as e:
        print(f"ExifTool check failed with error: {e}")

    # 2. Heuristic Fallback (Triggers if f_p is still None)
    if f_p is None:
        print(f"--- Metadata missing/invalid. Applying heuristic fallback for: {video_path} ---")
        # Standard CV heuristic: f = 1.2 * max(w, h)
        f_p = 1.2 * max(image_width, image_height)
        
    # 3. Build and Return Matrix
    cx = image_width / 2.0
    cy = image_height / 2.0
    
    K = np.array([
        [f_p, 0,  cx],
        [0,  f_p, cy],
        [0,  0,   1]
    ], dtype=np.float64)
    
    return K

def interpolate_paired_2d_3d_data(points_2d: dict, points_3d: dict, num_interpolated: int = 5) -> tuple[dict, dict]:
    """
    Interpolate additional 2D-3D point correspondences along tennis court lines.
    
    Args:
        points_2d: Dictionary mapping point names to 2D pixel coordinates
        points_3d: Dictionary mapping point names to 3D world coordinates
        num_interpolated: Number of points to interpolate between each pair
    
    Returns:
        interpolated_2d: Dictionary with original + interpolated 2D points
        interpolated_3d: Dictionary with original + interpolated 3D points
    """
    
    # Define the tennis court line segments to interpolate along
    # Based on standard tennis court layout
    court_lines = {
        'baseline_near': ['A', 'B'],      # Near baseline left segment
        'baseline_near2': ['B', 'C'],     # Near baseline middle segment  
        'baseline_near3': ['C', 'D'],     # Near baseline right segment
        'singles_left': ['B', 'E'],       # Left singles sideline
        'service_left': ['E', 'F'],       # Left service line
        'center_service': ['F', 'G'],     # Center service line
        'singles_right': ['C', 'G'],      # Right singles sideline
    }
    
    # Additional lines if more points are available
    extended_lines = {
        'service_mid_left': ['E', 'H'],   # Service box connections
        'net_left': ['F', 'I'],
        'service_mid_right': ['G', 'J'],
        'singles_mid_left': ['H', 'K'],
        'net_mid': ['I', 'L'],
        'singles_mid_right': ['J', 'M'],
        'baseline_far': ['N', 'O'],       # Far baseline
        'baseline_far2': ['O', 'P'],
        'baseline_far3': ['P', 'Q'],
        'singles_left_full': ['B', 'O'],  # Full court sidelines
        'singles_right_full': ['C', 'P'],
    }
    
    # Start with original points
    interpolated_2d = points_2d.copy()
    interpolated_3d = points_3d.copy()
    
    # Counter for naming interpolated points
    interp_counter = 0
    
    # Function to interpolate between two points
    def interpolate_between(p1_name, p2_name, n_points):
        nonlocal interp_counter
        
        if p1_name not in points_2d or p2_name not in points_2d:
            return  # Skip if points not available
        if p1_name not in points_3d or p2_name not in points_3d:
            return
            
        # Get 2D points
        p1_2d = np.array(points_2d[p1_name])
        p2_2d = np.array(points_2d[p2_name])
        
        # Get 3D points
        p1_3d = np.array(points_3d[p1_name])
        p2_3d = np.array(points_3d[p2_name])
        
        # Interpolate along the line segment
        for i in range(1, n_points + 1):
            t = i / (n_points + 1)
            
            # Linear interpolation in 2D
            interp_2d = (1 - t) * p1_2d + t * p2_2d
            
            # Linear interpolation in 3D
            interp_3d = (1 - t) * p1_3d + t * p2_3d
            
            # Create unique name for interpolated point
            interp_name = f"I_{p1_name}{p2_name}_{i}"
            
            interpolated_2d[interp_name] = tuple(interp_2d)
            interpolated_3d[interp_name] = tuple(interp_3d)
            
            interp_counter += 1
    
    # Interpolate along defined court lines
    for line_name, (start, end) in court_lines.items():
        interpolate_between(start, end, num_interpolated)
    
    # Try extended lines if those points exist
    for line_name, (start, end) in extended_lines.items():
        interpolate_between(start, end, num_interpolated)
    
    print(f"Added {interp_counter} interpolated points to {len(points_2d)} original points")
    print(f"Total points for calibration: {len(interpolated_2d)}")
    
    return interpolated_2d, interpolated_3d

def annotate_court(image_path, points_to_annotate=None, output_dir=None):
    """
    Main function to annotate tennis court points.
    
    Args:
        image_path: Path to the tennis court image
        points_to_annotate: List of point names to annotate (default: ['A', 'B', 'C', 'D', 'E', 'F', 'G'])
    
    Returns:
        Path to the saved JSON file, or None if cancelled
    """
    annotator = CourtAnnotator(image_path)
    
    if points_to_annotate:
        annotator.points_to_annotate = points_to_annotate
    
    annotator.run(output_dir)
    
    # Return the path to the saved file if it exists
    base_name = Path(image_path).stem
    output_file = os.path.join(output_dir, f"{base_name}_court_annotations.json")
    if os.path.exists(output_file):
        return output_file
    return None
